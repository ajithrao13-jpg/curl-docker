-- =============================================================

-- SQL-driven unit tests for dbo.USP_DELTA_MASTER
-- =============================================================


-- @@TEST: TC01 - Master executes without error against a clean DB
-- @@DESCRIPTION: Smoke test. USP_DELTA_MASTER completes without throwing.
--                Let exceptions propagate; the framework reports them.

EXEC dbo.USP_DELTA_MASTER;

-- @@ASSERT: Reached this point means EXEC did not throw
SELECT 1;

-- @@END_TEST


-- @@TEST: TC02 - Master does not leak transactions
-- @@DESCRIPTION: After USP_DELTA_MASTER returns, @@TRANCOUNT must match
--                the pre-call value and the session must not be in a
--                doomed transaction state.

DECLARE @before INT = @@TRANCOUNT;

EXEC dbo.USP_DELTA_MASTER;

DECLARE @after        INT = @@TRANCOUNT;
DECLARE @xact_state   INT = XACT_STATE();

-- @@ASSERT: TRANCOUNT unchanged
SELECT CASE WHEN @before = @after THEN 1 ELSE 0 END;

-- @@ASSERT: Session not in a doomed/uncommittable transaction
SELECT CASE WHEN @xact_state = 0 THEN 1 ELSE 0 END;

-- @@END_TEST


-- @@TEST: TC03 - End-to-end: master delegates to USP_DELTA_FOLDERS
-- @@DESCRIPTION: Seed STAGING + STAGING_FOLDERS, invoke master, and verify
--                that the rows landed in TARGET_FOLDERS, that SYS_DELTA_LOG
--                recorded a SUCCESS entry for FOLDERS, and that rows_upserted
--                matches what was seeded.

DECLARE @staging_id BIGINT;
DECLARE @req_id     NVARCHAR(64) = CONCAT('UT-TC03-', CONVERT(varchar(36), NEWID()));

INSERT INTO dbo.STAGING (requestId) VALUES (@req_id);
SET @staging_id = SCOPE_IDENTITY();

INSERT INTO dbo.STAGING_FOLDERS
    (folderIfUnmodifiedSince, folderFolderID, folderStateCode,
     folderTypeCode, folderDescription, folderOwnerEntityId,
     staging_id, Is_Processed)
VALUES
    (NULL, 'TC03-F001', 'ACTIVE', 'TYPE-A', 'Test Folder A', 'ENTITY-001', @staging_id, 'N'),
    (NULL, 'TC03-F002', 'CLOSED', 'TYPE-B', 'Test Folder B', 'ENTITY-002', @staging_id, 'N');

-- Capture log baseline so prior runs do not pollute assertions
DECLARE @log_baseline BIGINT =
    ISNULL((SELECT MAX(log_id) FROM dbo.SYS_DELTA_LOG), 0);

EXEC dbo.USP_DELTA_MASTER;

-- @@ASSERT: Both seeded staging rows landed in TARGET_FOLDERS (scoped by staging_id)
SELECT CASE WHEN COUNT(*) = 2 THEN 1 ELSE 0 END
FROM dbo.TARGET_FOLDERS
WHERE staging_id = @staging_id;

-- @@ASSERT: TARGET_FOLDERS contains the specific FolderIDs we seeded
SELECT CASE
    WHEN EXISTS (SELECT 1 FROM dbo.TARGET_FOLDERS
                 WHERE staging_id = @staging_id AND folderFolderID = 'TC03-F001')
     AND EXISTS (SELECT 1 FROM dbo.TARGET_FOLDERS
                 WHERE staging_id = @staging_id AND folderFolderID = 'TC03-F002')
    THEN 1 ELSE 0
END;

-- @@ASSERT: Latest FOLDERS log entry from this run is SUCCESS
SELECT CASE WHEN status = 'SUCCESS' THEN 1 ELSE 0 END
FROM dbo.SYS_DELTA_LOG
WHERE entity_name = 'FOLDERS'
  AND log_id > @log_baseline
ORDER BY log_id DESC
OFFSET 0 ROWS FETCH NEXT 1 ROW ONLY;

-- @@ASSERT: rows_upserted is 2
SELECT CASE WHEN rows_upserted = 2 THEN 1 ELSE 0 END
FROM dbo.SYS_DELTA_LOG
WHERE entity_name = 'FOLDERS'
  AND log_id > @log_baseline
ORDER BY log_id DESC
OFFSET 0 ROWS FETCH NEXT 1 ROW ONLY;

-- @@ASSERT: Seeded staging rows are flipped to processed
SELECT CASE WHEN COUNT(*) = 2 THEN 1 ELSE 0 END
FROM dbo.STAGING_FOLDERS
WHERE staging_id = @staging_id AND Is_Processed = 'Y';

-- @@END_TEST


-- @@TEST: TC04 - Empty staging produces clean no-op
-- @@DESCRIPTION: With no staging rows for FOLDERS, master must still complete
--                and log a SUCCESS entry with rows_upserted = 0.

DECLARE @log_baseline BIGINT =
    ISNULL((SELECT MAX(log_id) FROM dbo.SYS_DELTA_LOG), 0);

EXEC dbo.USP_DELTA_MASTER;

-- @@ASSERT: FOLDERS logged SUCCESS for this run
SELECT CASE WHEN status = 'SUCCESS' THEN 1 ELSE 0 END
FROM dbo.SYS_DELTA_LOG
WHERE entity_name = 'FOLDERS'
  AND log_id > @log_baseline
ORDER BY log_id DESC
OFFSET 0 ROWS FETCH NEXT 1 ROW ONLY;

-- @@ASSERT: rows_upserted is 0
SELECT CASE WHEN ISNULL(rows_upserted, 0) = 0 THEN 1 ELSE 0 END
FROM dbo.SYS_DELTA_LOG
WHERE entity_name = 'FOLDERS'
  AND log_id > @log_baseline
ORDER BY log_id DESC
OFFSET 0 ROWS FETCH NEXT 1 ROW ONLY;

-- @@END_TEST


-- @@TEST: TC05 - Idempotent: repeated execution does not duplicate target rows
-- @@DESCRIPTION: Seed two staging rows, run master three times, and verify
--                TARGET_FOLDERS still contains exactly 2 rows for this staging_id
--                (not 6). Confirms the hash-registry / dedup path holds.

DECLARE @staging_id BIGINT;
DECLARE @req_id     NVARCHAR(64) = CONCAT('UT-TC05-', CONVERT(varchar(36), NEWID()));

INSERT INTO dbo.STAGING (requestId) VALUES (@req_id);
SET @staging_id = SCOPE_IDENTITY();

INSERT INTO dbo.STAGING_FOLDERS
    (folderIfUnmodifiedSince, folderFolderID, folderStateCode,
     folderTypeCode, folderDescription, folderOwnerEntityId,
     staging_id, Is_Processed)
VALUES
    (NULL, 'TC05-F001', 'ACTIVE', 'TYPE-A', 'Idem Folder A', 'ENTITY-001', @staging_id, 'N'),
    (NULL, 'TC05-F002', 'ACTIVE', 'TYPE-A', 'Idem Folder B', 'ENTITY-002', @staging_id, 'N');

EXEC dbo.USP_DELTA_MASTER;
EXEC dbo.USP_DELTA_MASTER;
EXEC dbo.USP_DELTA_MASTER;

-- @@ASSERT: TARGET_FOLDERS still has exactly 2 rows for this staging_id
SELECT CASE WHEN COUNT(*) = 2 THEN 1 ELSE 0 END
FROM dbo.TARGET_FOLDERS
WHERE staging_id = @staging_id;

-- @@END_TEST


-- @@TEST: TC06 - Broken child SP does not stop other children
-- @@DESCRIPTION: Force USP_DELTA_ACCESSCONTROLGRANTS to throw, seed FOLDERS
--                staging, then run master. FOLDERS must still land in target
--                and log SUCCESS, proving the master continues past a failure.
--                A test-side CATCH guarantees the original SP is restored even
--                if the assertions blow up mid-test.

DECLARE @original_definition NVARCHAR(MAX) =
    OBJECT_DEFINITION(OBJECT_ID('dbo.USP_DELTA_ACCESSCONTROLGRANTS'));

DECLARE @staging_id BIGINT;
DECLARE @req_id     NVARCHAR(64) = CONCAT('UT-TC06-', CONVERT(varchar(36), NEWID()));
DECLARE @log_baseline BIGINT =
    ISNULL((SELECT MAX(log_id) FROM dbo.SYS_DELTA_LOG), 0);

BEGIN TRY

    -- Replace child SP with one that always throws
    EXEC('
        ALTER PROCEDURE dbo.USP_DELTA_ACCESSCONTROLGRANTS
        AS
        BEGIN
            THROW 50000, ''TC06 forced failure'', 1;
        END
    ');

    -- Seed FOLDERS so we have something downstream to verify
    INSERT INTO dbo.STAGING (requestId) VALUES (@req_id);
    SET @staging_id = SCOPE_IDENTITY();

    INSERT INTO dbo.STAGING_FOLDERS
        (folderIfUnmodifiedSince, folderFolderID, folderStateCode,
         folderTypeCode, folderDescription, folderOwnerEntityId,
         staging_id, Is_Processed)
    VALUES
        (NULL, 'TC06-F001', 'ACTIVE', 'TYPE-A', 'After-failure Folder', 'ENTITY-001', @staging_id, 'N');

    EXEC dbo.USP_DELTA_MASTER;

END TRY
BEGIN CATCH
    -- Swallow so the restore below always runs
    PRINT 'TC06 body raised: ' + ERROR_MESSAGE();
END CATCH;

-- Always restore, even if the master or assertions threw
IF @original_definition IS NOT NULL
    EXEC(@original_definition);

-- @@ASSERT: FOLDERS row reached TARGET_FOLDERS despite the broken child
SELECT CASE WHEN COUNT(*) = 1 THEN 1 ELSE 0 END
FROM dbo.TARGET_FOLDERS
WHERE staging_id = @staging_id
  AND folderFolderID = 'TC06-F001';

-- @@ASSERT: FOLDERS logged SUCCESS for this run
--           (Once master logs failures, also add an assertion that
--            ACCESSCONTROLGRANTS has a FAILED row > @log_baseline.)
SELECT CASE WHEN status = 'SUCCESS' THEN 1 ELSE 0 END
FROM dbo.SYS_DELTA_LOG
WHERE entity_name = 'FOLDERS'
  AND log_id > @log_baseline
ORDER BY log_id DESC
OFFSET 0 ROWS FETCH NEXT 1 ROW ONLY;

-- @@END_TEST
